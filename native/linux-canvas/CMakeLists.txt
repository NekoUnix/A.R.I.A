cmake_minimum_required(VERSION 3.16)
project(aria_obs_canvas C)
set(CMAKE_C_STANDARD 11)
find_package(PkgConfig REQUIRED)
pkg_check_modules(OBS REQUIRED IMPORTED_TARGET libobs)
# Some distributions' libobs.pc omits the obs/ include subdirectory.
find_path(OBS_HEADER_DIR obs-module.h HINTS ${OBS_INCLUDE_DIRS} PATH_SUFFIXES obs)
if(NOT OBS_HEADER_DIR)
    message(FATAL_ERROR "Install the libobs development headers for your OBS build")
endif()
add_library(aria-canvas MODULE obs-plugin.c transport.c)
target_link_libraries(aria-canvas PRIVATE PkgConfig::OBS pthread)
target_include_directories(aria-canvas SYSTEM PRIVATE ${OBS_HEADER_DIR})
target_compile_options(aria-canvas PRIVATE -Wall -Wextra -Werror)
set_target_properties(aria-canvas PROPERTIES PREFIX "")
install(TARGETS aria-canvas LIBRARY DESTINATION lib/obs-plugins)
enable_testing()
add_executable(canvas-test transport-test.c transport.c)
target_compile_options(canvas-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME canvas-protocol COMMAND canvas-test)
add_executable(obs-canvas-test obs-test.c transport.c)
target_link_libraries(obs-canvas-test PRIVATE PkgConfig::OBS)
target_include_directories(obs-canvas-test SYSTEM PRIVATE ${OBS_HEADER_DIR})
target_compile_options(obs-canvas-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
