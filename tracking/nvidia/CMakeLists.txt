cmake_minimum_required(VERSION 3.21)
project(aria_nvar LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The ARIA NVIDIA bridge currently targets Windows x64.")
endif()
set(ARSDK_ROOT "" CACHE PATH "Extracted NVIDIA AR SDK Core with installed FaceExpressions, LandmarkDetection and FaceBoxDetection features")
if(NOT IS_DIRECTORY "${ARSDK_ROOT}")
    message(FATAL_ERROR "Set -DARSDK_ROOT to your NVIDIA AR SDK directory. See docs/webcam.md.")
endif()
file(GLOB_RECURSE AR_HEADERS "${ARSDK_ROOT}/*.h")
set(AR_INCLUDE_DIRS "")
foreach(HEADER IN LISTS AR_HEADERS)
    get_filename_component(DIR "${HEADER}" DIRECTORY)
    list(APPEND AR_INCLUDE_DIRS "${DIR}")
endforeach()
list(REMOVE_DUPLICATES AR_INCLUDE_DIRS)
foreach(NAME nvAR.h nvARFaceExpressions.h nvARLandmarkDetection.h nvARFaceBoxDetection.h nvCVImage.h)
    find_file(FOUND_${NAME} "${NAME}" PATHS ${AR_INCLUDE_DIRS} NO_DEFAULT_PATH REQUIRED)
endforeach()
add_library(aria-nvar SHARED bridge.cpp)
target_compile_features(aria-nvar PRIVATE cxx_std_17)
target_include_directories(aria-nvar PRIVATE ${AR_INCLUDE_DIRS})
file(GLOB_RECURSE AR_LIBRARIES "${ARSDK_ROOT}/*.lib")
foreach(NAME nvARPose NVCVImage)
    set(MATCHING "")
    foreach(LIB IN LISTS AR_LIBRARIES)
        get_filename_component(STEM "${LIB}" NAME_WE)
        string(TOLOWER "${STEM}" STEM)
        string(TOLOWER "${NAME}" LOWER_NAME)
        if(STEM STREQUAL LOWER_NAME AND NOT LIB MATCHES "[Aa][Rr][Mm]64|[Ww]in32|/x86/")
            list(APPEND MATCHING "${LIB}")
        endif()
    endforeach()
    list(LENGTH MATCHING COUNT)
    if(COUNT EQUAL 0)
        # SDK packages may supply NVIDIA's proxy sources instead of import libraries.
        if(NAME STREQUAL "nvARPose")
            set(PROXY "${ARSDK_ROOT}/nvar/src/nvARProxy.cpp")
        else()
            set(PROXY "${ARSDK_ROOT}/nvar/src/nvCVImageProxy.cpp")
        endif()
        if(NOT EXISTS "${PROXY}")
            message(FATAL_ERROR "Missing ${NAME} import library and SDK proxy source.")
        endif()
        target_sources(aria-nvar PRIVATE "${PROXY}")
    elseif(NOT COUNT EQUAL 1)
        message(FATAL_ERROR "Expected one x64 ${NAME}.lib, found ${COUNT}. Use an SDK containing one matching version of each feature.")
    else()
        target_link_libraries(aria-nvar PRIVATE ${MATCHING})
    endif()
endforeach()
# Features are loaded by nvARPose at runtime; their SDK targets are header-only
# interfaces. Requiring feature .lib files would reject valid NVIDIA packages.
install(TARGETS aria-nvar RUNTIME DESTINATION .)
